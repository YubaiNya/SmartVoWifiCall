cmake_minimum_required(VERSION 3.22)
project(smartvowifi_lpa LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
file(GLOB EUICC CONFIGURE_DEPENDS lpa/lpac/euicc/*.c)
file(GLOB JSON_EXT CONFIGURE_DEPENDS lpa/lpac/cjson-ext/cjson-ext/*.c)
file(GLOB JNI_BRIDGE CONFIGURE_DEPENDS lpa/lpac-jni/*.c)
add_library(lpac-jni SHARED ${EUICC} ${JSON_EXT} ${JNI_BRIDGE} lpa/cjson/cjson/cJSON.c)
target_include_directories(lpac-jni PRIVATE lpa/lpac lpa/lpac/cjson-ext lpa/cjson)
target_compile_options(lpac-jni PRIVATE -fvisibility=hidden -ffile-prefix-map=${CMAKE_SOURCE_DIR}=lpa)
target_link_options(lpac-jni PRIVATE -Wl,-z,max-page-size=16384)
